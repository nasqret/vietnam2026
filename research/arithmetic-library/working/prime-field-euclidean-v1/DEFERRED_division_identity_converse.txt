    second = ('QB', 'QC', 'Q', 'PB', 'PC', 'UB', 'UC', 'T', 'RB', 'RC', 'K')
    params = (*initial, *first, *second)
    body = _intro(*params, 'hprime', 'hdivisor', 'hcanonical', 'hother_canonical', 'hlength', 'hother_length',
                  'hresidual', 'hother_residual', 'hbound', 'hother_bound') + _parts('hdivisor', 3)
    body += ('cases hdivisor_right_right', 'cases hdivisor_right_right_witness')
    body += (f"have hinverse : exists k. ({_inv('p', 'x', 'k', 'aligned_identity_inverse')})",)
    body += _call('prime_field_inverse_exists', 'p', 'x') + ('exact hprime',)

    body += ('exact hdivisor_right_left', 'exists d', 'simp', 'exact hdivisor_right_right_witness_left',
             'exact hdivisor_right_right_witness_right', 'cases hinverse')

                  'x', 'x1', 'q', 'qb', 'qc', 'x', 'x1', 'Q', 'QB', 'QC')

        (first, 'hcanonical', 'hlength', 'hresidual', 'hbound'),
        (second, 'hother_canonical', 'hother_length', 'hother_residual', 'hother_bound'),
